SELECT TRIGGER FROM
  Blue Down => Coords, Decrease, Idle;
  Yellow Down WHILE LeftShift Down => Coords, Reset, Idle;
  Yellow Down => Coords, Random, Idle;
  Red Down => Coords, Increase, Idle;
  Mouse => SELECT ENABLE FROM
  	Blue Down => Coords, Decrease, Idle;
  	Yellow Down => SELECT ENABLE FROM
  		LeftShift Down => Coords, Reset, Idle;
  		ENDCASE => Coords, Random, Idle;
  	Red Down => Coords, Increase, Idle;
  	ENDCASE => Coords;
  Blue Up => Coords, Decrease, Doit;
  Yellow Up WHILE LeftShift Down => Coords, Reset, Doit;
  Yellow Up => Coords, Random, Doit;
  Red Up => Coords, Increase, Doit;
  ENDCASE.
