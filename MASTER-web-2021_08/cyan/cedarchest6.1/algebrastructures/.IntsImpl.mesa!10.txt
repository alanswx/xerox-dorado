Read: PUBLIC AC.ReadOp ~ {
    RETURN[NEW[AC.ObjectRec _ [
        structure: Ints,
        data: NEW[INT  _ IO.GetInt[in] ]
        ] ] ];
    };

    FromRope: PUBLIC AC.FromRopeOp = {
        stream: IO.STREAM _ IO.RIS[in];
        RETURN[ Read[stream, structure] ];
        };

    ToRope: PUBLIC AC.ToRopeOp = {
        data: IntData _ NARROW[in.data];
        RETURN[ Convert.RopeFromInt[data^] ];
        };

Write: PUBLIC AC.WriteOp = {
IO.PutRope[ stream, ToRope[in] ]
}; 

    FromINT: PUBLIC PROC [int: INT] RETURNS [Int] = {
        RETURN[NEW[AC.ObjectRec _ [
            structure: Ints,
            data: NEW[INT  _ int]
            ] ] ];
        };

    ToINT: PUBLIC PROC [int: Int] RETURNS [INT] = {
        data: IntData _ NARROW[int.data];
        RETURN[data^];
        };
<<Arithmetic>>
Add: PUBLIC AC.BinaryOp ~ {
    firstData: IntData _ NARROW[firstArg.data];
    secondData: IntData _ NARROW[secondArg.data];
    RETURN[NEW[AC.ObjectRec _ [
        structure: Ints,
        data: NEW[INT  _ firstData^ + secondData^  ]
        ] ] ];
    };

Negate: PUBLIC AC.UnaryOp ~ {
    data: IntData _ NARROW[arg.data];
    RETURN[NEW[AC.ObjectRec _ [
        structure: Ints,
        data: NEW[INT  _ - data^  ]
        ] ] ];
    };

Subtract: PUBLIC AC.BinaryOp ~ {
    firstData: IntData _ NARROW[firstArg.data];
    secondData: IntData _ NARROW[secondArg.data];
    RETURN[NEW[AC.ObjectRec _ [
        structure: Ints,
        data: NEW[INT  _ firstData^ - secondData^  ]

    };

Multiply: PUBLIC AC.BinaryOp ~ {
    firstData: IntData _ NARROW[firstArg.data];
    secondData: IntData _ NARROW[secondArg.data];
    RETURN[NEW[AC.ObjectRec _ [
        structure: Ints,

Remainder: PUBLIC AC.BinaryOp ~ {
    firstData: IntData _ NARROW[firstArg.data];
    secondData: IntData _ NARROW[secondArg.data];
    RETURN[NEW[AC.ObjectRec _ [
        structure: Ints,
        data: NEW[INT  _ firstData^ MOD secondData^  ]
        ] ] ];
    };

Gcd: PUBLIC AC.BinaryOp ~ {
gcd: Int;
IF Equal[firstArg, IntZero] AND Equal[secondArg, IntZero] THEN gcd _ IntZero
ELSE {
r: Int;
UNTIL Equal[secondArg, IntZero] DO
r _ Remainder[firstArg, secondArg];
firstArg _ secondArg;  secondArg _ r;
ENDLOOP;
gcd _ Abs[firstArg];
};
    RETURN[gcd ];
    };
<<Comparison>>
Sign:  PUBLIC AC.CompareToZeroOp = {
    data: IntData _ NARROW[arg.data];
    SELECT data^ FROM
        < 0 => RETURN[less];
        = 0 => RETURN[equal];
        ENDCASE => RETURN[greater];
};
<<>>
Abs: PUBLIC AC.UnaryOp ~ {
    data: IntData _ NARROW[arg.data];
    RETURN[NEW[AC.ObjectRec _ [
        structure: Ints,
        data: NEW[INT  _ ABS[data^] ]
        ] ] ];
};

Compare: PUBLIC AC.BinaryCompareOp ~ {
    firstData: IntData _ NARROW[firstArg.data];
    secondData: IntData _ NARROW[secondArg.data];
    SELECT firstData^ FROM
        < secondData^ => RETURN[less];
        = secondData^ => RETURN[equal];
        ENDCASE => RETURN[greater];
};

Equal: PUBLIC AC.EqualityOp ~ {
    firstData: IntData _ NARROW[firstArg.data];
    secondData: IntData _ NARROW[secondArg.data];
RETURN[ firstData^ = secondData^]
};
