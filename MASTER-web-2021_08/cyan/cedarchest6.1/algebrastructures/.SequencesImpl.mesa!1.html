IMPORTS Rope, Atom, IO, Convert, MathConstructors, AlgebraClasses, Ints
EXPORTS Sequences =
BEGIN OPEN AC: AlgebraClasses, Sequences;
Errors
SyntaxError: PUBLIC ERROR [reason: ATOM] = CODE;
BadElementStructure: PUBLIC ERROR [elementStructure: AC.Structure] = CODE;
TypeError: PUBLIC ERROR [message: ATOM ← $Unspecified] = CODE;
Class for Sequence Structures
ClassPrintName: AC.PrintNameProc = {
data: SequenceStructureData ← NARROW[structure.instanceData];
RETURN[Rope.Concat[
"Sequences over ",
data.elementStructure.class.printName[data.elementStructure]
] ];
};
ClassShortPrintName: AC.PrintNameProc = {
data: SequenceStructureData ← NARROW[structure.instanceData];
RETURN[Rope.Cat[
"S(",
data.elementStructure.class.shortPrintName[data.elementStructure],
")"
] ];
};
ClassReportOps: AC.ReportOpsProc = {
opNames ← CONS["ApplyUnaryOp", opNames];
refOps ← CONS[NEW[AC.BinaryMixedOp ← ApplyUnOp ], refOps];
opNames ← CONS["Delete", opNames];
refOps ← CONS[NEW[AC.BinaryOp ← Del ], refOps];
opNames ← CONS["Insert", opNames];
refOps ← CONS[NEW[AC.TernaryOp ← Ins ], refOps];
opNames ← CONS["Select", opNames];
refOps ← CONS[NEW[AC.BinaryOp ← Sel ], refOps];
opNames ← CONS["FromLinearRope", opNames];
refOps ← CONS[NEW[AC.FromRopeOp ← FromRope], refOps];
};
ClassIsElementOf: AC.ElementOfProc = {
Assumes that if item is a sequence, then it really belongs to the domain pointed to by its structure field.
IF NOT structure.class.structureEqual[structure, item.structure] THEN RETURN[FALSE];
RETURN[ TRUE ]
};
ClassLegalFirstChar: AC.LegalFirstCharOp = {
SELECT char FROM
'[ , '(=> RETURN[TRUE];
ENDCASE;
RETURN[FALSE];
};
ClassToExpr: AC.ToExprOp = {
data: SequenceStructureData ← NARROW[in.structure.instanceData];
inData: SequenceData ← NARROW[in.data];
outColumn: LIST OF MathExpr.EXPR ← NIL;
size: CARDINAL ← inData.lengthPlus1 - 1;
IF size < 10 THEN {
FOR j:NAT DECREASING IN [1..size] DO
outColumn ← CONS[data.elementStructure.class.toExpr[inData[j] ], outColumn];
ENDLOOP;
out ← MathConstructors.MakeVector[size, outColumn, FALSE];
}
ELSE
out ← MathConstructors.MakeVariable[Rope.Concat[Convert.RopeFromCard[size], " items"] ];
};
Sequence Structure Constructor
MakeSequenceStructure: PUBLIC PROC [elementStructure: AC.Structure, firstNullaryOpName: Rope.ROPE ← NIL, firstNullaryOp: AC.NullaryOp ← NIL, firstUnaryOpName: Rope.ROPE ← NIL, firstUnaryOp: AC.UnaryOp ← NIL, secondNullaryOpName: Rope.ROPE ← NIL, secondNullaryOp: AC.NullaryOp ← NIL, secondUnaryOpName: Rope.ROPE ← NIL, secondUnaryOp: AC.UnaryOp ← NIL] RETURNS [sequenceStructure: AC.Structure] ~ {
sequenceStructureData: SequenceStructureData ← NEW[SequenceStructureDataRec ← [
elementStructure: elementStructure
] ];
sequenceStructureOps: SequenceOps ← NEW[SequenceOpsRec ← [
select: Sel,
insert: Ins,
delete: Del,
applyUnaryOpToElements: ApplyUnOp,
makeSequence: MakeSeq,
firstNullaryOpName: firstNullaryOpName,
firstNullaryOp: firstNullaryOp,
secondNullaryOpName: secondNullaryOpName,
secondNullaryOp: secondNullaryOp,
firstUnaryOpName: firstUnaryOpName,
firstUnaryOp: firstUnaryOp,
secondUnaryOpName: secondUnaryOpName,
secondUnaryOp: secondUnaryOp
] ];
sequenceProp: Atom.DottedPair ← NEW[Atom.DottedPairNode← [$SequenceStructure, sequenceStructureOps]];
sequenceStructureClass: AC.StructureClass ← NEW[AC.StructureClassRec ← [
category: set,
printName: ClassPrintName,
shortPrintName: ClassShortPrintName,
structureEqual: AC.defaultStructureEqualityTest,
reportOps: ClassReportOps,
isElementOf: ClassIsElementOf,
legalFirstChar: ClassLegalFirstChar,
read: Read,
fromRope: FromRope,
toRope: ToRope,
write: Write,
toExpr: ClassToExpr,
equal: Equal,
integralDomain: FALSE,
gcdDomain: FALSE,
euclideanDomain: FALSE,
propList: LIST[sequenceProp]
] ];
RETURN[ NEW[AC.StructureRec ← [
class: sequenceStructureClass,
instanceData: sequenceStructureData
] ] ];
};
Extract Sequence Operations from Class Record Property Lists
IsSequenceStructure: PUBLIC PROC [structure: AC.Structure] RETURNS [BOOL] ~ {
IF Atom.GetPropFromList[structure.class.propList, $SequenceStructure] # NIL THEN RETURN[TRUE] ELSE RETURN[FALSE];
};
Select: PUBLIC PROC [structure: AC.Structure] RETURNS [AC.BinaryOp] ~ {
IF IsSequenceStructure[structure] THEN {
sequenceOps: SequenceOps ← NARROW[ Atom.GetPropFromList[structure.class.propList, $SequenceStructure] ];
RETURN[sequenceOps.select];
}
ELSE ERROR;
};
Insert: PUBLIC PROC [structure: AC.Structure] RETURNS [AC.TernaryOp] ~ {
IF IsSequenceStructure[structure] THEN {
sequenceOps: SequenceOps ← NARROW[ Atom.GetPropFromList[structure.class.propList, $SequenceStructure] ];
RETURN[sequenceOps.insert];
}
ELSE ERROR;
};
Delete: PUBLIC PROC [structure: AC.Structure] RETURNS [AC.BinaryOp] ~ {
IF IsSequenceStructure[structure] THEN {
sequenceOps: SequenceOps ← NARROW[ Atom.GetPropFromList[structure.class.propList, $SequenceStructure] ];
RETURN[sequenceOps.delete];
}
ELSE ERROR;
};
ApplyUnaryOpToElements: PUBLIC PROC [structure: AC.Structure] RETURNS [AC.BinaryMixedOp] ~ {

MakeSequence: PUBLIC PROC [structure: AC.Structure] RETURNS [AC.ListImbedOp] ~ {

Constructor
MakeSeq: PUBLIC AC.ListImbedOp ~ {
structureData: SequenceStructureData ← NARROW[structure.instanceData];
elementStructure: AC.Structure ← structureData.elementStructure;
length: NAT ← GList.Length[data];
length: NAT ← 0;
ptr: LIST OF AC.Object ← data;
outData: SequenceData;
WHILE ptr#NIL DO length ← length + 1; ptr ← ptr.rest ENDLOOP;
outData ← NEW[SequenceDataRec[length] ];
FOR i:NAT IN [1..length] DO
outData[i] ← data.first;
data ← data.rest;
ENDLOOP;
out ← NEW[AC.ObjectRec ← [structure: structure, data: outData] ];
};
Conversion and IO
Read: PUBLIC AC.ReadOp ~ {
structureData: SequenceStructureData ← NARROW[structure.instanceData];
elementStructure: AC.Structure ← structureData.elementStructure;
puncChar: CHAR;
nextElement: AC.Object;
length: NAT ← 0;
ReadFail: PUBLIC ERROR [subclass: ATOM ← $Unspecified] = CODE;
list, listTail: LIST OF AC.Object ← NIL;
outData: SequenceData;
[]← in.SkipWhitespace[];
puncChar ← in.GetChar[];
[]← in.SkipWhitespace[];
IF puncChar # '( AND puncChar # '[ THEN ReadFail[$LeftParenExpected];
[]← in.SkipWhitespace[];
puncChar ← in.PeekChar[];
IF puncChar = ') OR puncChar = '] THEN puncChar ← in.GetChar[];
WHILE puncChar # ') AND puncChar # '] DO
nextElement ← elementStructure.class.read[in, elementStructure];
length ← length + 1;
[]← in.SkipWhitespace[];
IF list=NIL THEN list ← listTail ←LIST[nextElement] ELSE
{ listTail.rest ← LIST[nextElement]; listTail ← listTail.rest };
puncChar ← in.GetChar[];
[]← in.SkipWhitespace[];
IF puncChar # ') AND puncChar # '] THEN
IF puncChar # ', THEN ReadFail[$CommaExpected];
ENDLOOP;
outData ← NEW[SequenceDataRec[length] ];
FOR i:NAT IN [1..length] DO
outData[i] ← list.first;
list ← list.rest;
ENDLOOP;
out ← NEW[AC.ObjectRec ← [structure: structure, data: outData] ];
};
FromRope: PUBLIC AC.FromRopeOp ~ {
out ← Read[IO.RIS[in], structure];
};
ToRope: PUBLIC AC.ToRopeOp ~ {
sequenceStructureData: SequenceStructureData ← NARROW[in.structure.instanceData];
elementStructure: AC.Structure ← sequenceStructureData.elementStructure;
inData: SequenceData ← NARROW[in.data];
out ← "( ";
FOR i:NAT IN [1..inData.lengthPlus1) DO
out ← Rope.Concat[ out, elementStructure.class.toRope[inData[i] ] ];
IF i < inData.lengthPlus1-1 THEN out ← Rope.Concat[out,", "];
ENDLOOP;
out ← Rope.Concat[ out, " )" ];
};
Write: PUBLIC AC.WriteOp ~ {
stream.PutRope[ ToRope[in] ]
};
Operations
Equal: PUBLIC AC.EqualityOp ~ {
data: SequenceStructureData ← NARROW[firstArg.structure.instanceData];
firstArgData: SequenceData ← NARROW[firstArg.data];
secondArgData: SequenceData ← NARROW[secondArg.data];
IF firstArgData.lengthPlus1 # secondArgData.lengthPlus1 THEN RETURN[FALSE];
FOR j: NAT IN [1..firstArgData.lengthPlus1) DO
IF NOT data.elementStructure.class.equal[firstArgData[j], secondArgData[j] ] THEN RETURN[FALSE];
ENDLOOP;
RETURN[TRUE];
};
Sel: PUBLIC AC.BinaryOp ~ {
firstData: SequenceData ← NARROW[firstArg.data];
index: INT ← Ints.ToINT[secondArg];
RETURN[firstData[index] ];
};
Ins: PUBLIC AC.TernaryOp ~ {
sequenceStructureData: SequenceStructureData ← NARROW[firstArg.structure.instanceData];
firstData: SequenceData ← NARROW[firstArg.data];
insertAfter: INT ← Ints.ToINT[secondArg];
newData: SequenceData ← NEW[SequenceDataRec[firstData.lengthPlus1] ];
IF NOT sequenceStructureData.elementStructure.class.isElementOf[thirdArg, sequenceStructureData.elementStructure] THEN TypeError[];
FOR i:INT IN [1..insertAfter] DO newData[i] ← firstData[i] ENDLOOP;
newData[insertAfter+1] ← thirdArg;
FOR j:INT IN [insertAfter+1..firstData.lengthPlus1) DO newData[j+1] ← firstData[j] ENDLOOP;
RETURN[ NEW[AC.ObjectRec ← [
structure: firstArg.structure,
data: newData
] ] ];
};
Del: PUBLIC AC.BinaryOp ~ {
sequenceStructureData: SequenceStructureData ← NARROW[firstArg.structure.instanceData];
firstData: SequenceData ← NARROW[firstArg.data];
delete: INT ← Ints.ToINT[secondArg];
newData: SequenceData ← NEW[SequenceDataRec[firstData.lengthPlus1-2] ];
FOR i:INT IN [1..delete-1] DO newData[i] ← firstData[i] ENDLOOP;
FOR j:INT IN [delete+1..firstData.lengthPlus1) DO newData[j-1] ← firstData[j] ENDLOOP;
RETURN[ NEW[AC.ObjectRec ← [
structure: firstArg.structure,
data: newData
] ] ];
};
ApplyUnOp: PUBLIC AC.BinaryMixedOp ~ {
seqData: SequenceData ← NARROW[firstArg.data];
refOp: REF AC.UnaryOp ← NARROW[secondArg];
newElementStructure, newSeqStructure: AC.Structure ← NIL;
newData: SequenceData ← NEW[SequenceDataRec[seqData.lengthPlus1-1] ];
FOR i:INT IN [1..seqData.lengthPlus1) DO
newData[i] ← refOp^[seqData[i]];
IF newData[i]# NIL THEN newElementStructure ← newData[i].structure;
ENDLOOP;
IF newElementStructure = NIL THEN RETURN[NIL];
newSeqStructure ← MakeSequenceStructure[newElementStructure];
RETURN[ NEW[AC.ObjectRec ← [
structure: newSeqStructure,
data: newData
] ] ];
};
END.
