<<Class for Sequence Structures>>
ClassPrintName: AC.PrintNameProc = {
data: SequenceStructureData _ NARROW[structure.instanceData];
RETURN[Rope.Concat[
"Sequences over ",
data.elementStructure.class.printName[data.elementStructure]
] ];
};
ClassShortPrintName: AC.PrintNameProc = {
data: SequenceStructureData _ NARROW[structure.instanceData];
RETURN[Rope.Cat[
"S(",
data.elementStructure.class.shortPrintName[data.elementStructure],
")"
] ];
};
ClassReportOps: AC.ReportOpsProc = {
    opNames _ CONS["ApplyUnaryOp", opNames];
    refOps _ CONS[NEW[AC.BinaryMixedOp _ ApplyUnOp ], refOps];
    opNames _ CONS["Delete", opNames];
    refOps _ CONS[NEW[AC.BinaryOp _ Del ], refOps];
    opNames _ CONS["Insert", opNames];
    refOps _ CONS[NEW[AC.TernaryOp _ Ins ], refOps];
    opNames _ CONS["Select", opNames];
    refOps _ CONS[NEW[AC.BinaryOp _ Sel ], refOps];
    opNames _ CONS["FromLinearRope", opNames];
    refOps _ CONS[NEW[AC.FromRopeOp _ FromRope], refOps];
}; 
ClassIsElementOf: AC.ElementOfProc = {
<<Assumes that if item is a sequence, then it really belongs to the domain pointed to by its structure field.>>
IF NOT structure.class.structureEqual[structure, item.structure] THEN RETURN[FALSE];
RETURN[ TRUE ]
}; 
ClassLegalFirstChar: AC.LegalFirstCharOp = {
    SELECT char FROM
        '[ , '(=> RETURN[TRUE];
        ENDCASE;
    RETURN[FALSE];
};
ClassToExpr: AC.ToExprOp = {
data: SequenceStructureData _ NARROW[in.structure.instanceData];
inData: SequenceData _ NARROW[in.data];
outColumn: LIST OF MathExpr.EXPR _ NIL;
size: CARDINAL _ inData.lengthPlus1 - 1;
IF size < 10 THEN {
FOR j:NAT DECREASING IN [1..size] DO
outColumn _ CONS[data.elementStructure.class.toExpr[inData[j] ], outColumn];
    ENDLOOP;
out _  MathConstructors.MakeVector[size, outColumn, FALSE];
}
ELSE 
out _ MathConstructors.MakeVariable[Rope.Concat[Convert.RopeFromCard[size], " items"] ];
}; 
<<Sequence Structure Constructor>>

    AC.NullaryOp _ NIL, firstUnaryOpName: Rope.ROPE _ NIL, firstUnaryOp: AC.UnaryOp _ NIL, secondNullaryOpName: Rope.ROPE _ NIL, 
    secondNullaryOp: AC.NullaryOp _ NIL, secondUnaryOpName: Rope.ROPE _ NIL, secondUnaryOp: AC.UnaryOp _ NIL] RETURNS 
    [sequenceStructure: AC.Structure] ~ {
        sequenceStructureData: SequenceStructureData _ NEW[SequenceStructureDataRec _ [
            elementStructure: elementStructure
            ] ];
        sequenceStructureOps: SequenceOps _ NEW[SequenceOpsRec _ [
            select: Sel,
            insert: Ins,
            delete: Del,
            applyUnaryOpToElements: ApplyUnOp, 
            makeSequence: MakeSeq,
            firstNullaryOpName: firstNullaryOpName,
            firstNullaryOp: firstNullaryOp,
            secondNullaryOpName: secondNullaryOpName,
            secondNullaryOp: secondNullaryOp,
            firstUnaryOpName: firstUnaryOpName,
            firstUnaryOp: firstUnaryOp,
            secondUnaryOpName: secondUnaryOpName,
            secondUnaryOp: secondUnaryOp
            ] ];
        sequenceProp: Atom.DottedPair _ NEW[Atom.DottedPairNode_ [$SequenceStructure, sequenceStructureOps]];
    sequenceStructureClass: AC.StructureClass _ NEW[AC.StructureClassRec _ [
    category: set,
    printName: ClassPrintName,
    shortPrintName: ClassShortPrintName,

<<Extract Sequence Operations from Class Record Property Lists>>
    IsSequenceStructure: PUBLIC PROC [structure: AC.Structure] RETURNS [BOOL] ~ {
        IF Atom.GetPropFromList[structure.class.propList, $SequenceStructure] # NIL THEN RETURN[TRUE] ELSE RETURN[FALSE];
        };

    Select: PUBLIC PROC [structure: AC.Structure] RETURNS [AC.BinaryOp] ~ {
        IF IsSequenceStructure[structure] THEN {
            sequenceOps: SequenceOps  _ NARROW[ Atom.GetPropFromList[structure.class.propList, $SequenceStructure] ];
            RETURN[sequenceOps.select];
            }
        ELSE ERROR;
        };

    Insert: PUBLIC PROC [structure: AC.Structure] RETURNS [AC.TernaryOp] ~ {
        IF IsSequenceStructure[structure] THEN {
            sequenceOps: SequenceOps  _ NARROW[ Atom.GetPropFromList[structure.class.propList, $SequenceStructure] ];
            RETURN[sequenceOps.insert];
            }
        ELSE ERROR;
        };

    Delete: PUBLIC PROC [structure: AC.Structure] RETURNS [AC.BinaryOp] ~ {
        IF IsSequenceStructure[structure] THEN {
            sequenceOps: SequenceOps  _ NARROW[ Atom.GetPropFromList[structure.class.propList, $SequenceStructure] ];
            RETURN[sequenceOps.delete];
            }
        ELSE ERROR;
        };

    ApplyUnaryOpToElements: PUBLIC PROC [structure: AC.Structure] RETURNS [AC.BinaryMixedOp] ~ {
        IF IsSequenceStructure[structure] THEN {
            sequenceOps: SequenceOps  _ NARROW[ Atom.GetPropFromList[structure.class.propList, $SequenceStructure] ];
            RETURN[sequenceOps.applyUnaryOpToElements];
            }
        ELSE ERROR;
        };

    MakeSequence: PUBLIC PROC [structure: AC.Structure] RETURNS [AC.ListImbedOp] ~ {
        IF IsSequenceStructure[structure] THEN {
            sequenceOps: SequenceOps  _ NARROW[ Atom.GetPropFromList[structure.class.propList, $SequenceStructure] ];
            RETURN[sequenceOps.makeSequence];
            }
        ELSE ERROR;
        };

    FirstNullaryOp: PUBLIC PROC [structure: AC.Structure] RETURNS [opName: Rope.ROPE, op: AC.NullaryOp] ~ {

<<Constructor>>
MakeSeq: PUBLIC AC.ListImbedOp ~ {
structureData: SequenceStructureData _ NARROW[structure.instanceData];
elementStructure: AC.Structure _ structureData.elementStructure;
    <<length: NAT _ GList.Length[data];>>
    length: NAT _ 0;
    ptr: LIST OF AC.Object _ data;
outData: SequenceData;
WHILE ptr#NIL DO length _ length + 1; ptr _ ptr.rest ENDLOOP;
outData _ NEW[SequenceDataRec[length] ];
FOR i:NAT IN [1..length] DO
outData[i] _ data.first;
data _ data.rest;
ENDLOOP;
out _ NEW[AC.ObjectRec _ [structure: structure, data: outData] ];
};
<<Conversion and IO>>
Read: PUBLIC AC.ReadOp ~ {
structureData: SequenceStructureData _ NARROW[structure.instanceData];
elementStructure: AC.Structure _ structureData.elementStructure;
puncChar: CHAR;
    nextElement: AC.Object;
    length: NAT _ 0;
ReadFail: PUBLIC ERROR [subclass: ATOM _ $Unspecified] = CODE;
list, listTail: LIST OF AC.Object _ NIL;
outData: SequenceData;
[]_ in.SkipWhitespace[];
puncChar _ in.GetChar[];
[]_ in.SkipWhitespace[];
IF puncChar # '( AND puncChar # '[ THEN ReadFail[$LeftParenExpected];
[]_ in.SkipWhitespace[];
puncChar _ in.PeekChar[];
IF puncChar = ') OR puncChar = '] THEN puncChar _ in.GetChar[];
WHILE puncChar # ') AND puncChar # '] DO
nextElement _ elementStructure.class.read[in, elementStructure];
    length _ length + 1;
[]_ in.SkipWhitespace[];
IF list=NIL THEN list _ listTail _LIST[nextElement] ELSE
{ listTail.rest _ LIST[nextElement];  listTail _ listTail.rest };
puncChar _ in.GetChar[];
[]_ in.SkipWhitespace[];
IF puncChar # ') AND puncChar # '] THEN
IF puncChar # ', THEN ReadFail[$CommaExpected];
    ENDLOOP;
outData _ NEW[SequenceDataRec[length] ];
FOR i:NAT IN [1..length] DO
outData[i] _ list.first;
list _ list.rest;
ENDLOOP;
out _ NEW[AC.ObjectRec _ [structure: structure, data: outData] ];
};

FromRope: PUBLIC AC.FromRopeOp ~ {
out _ Read[IO.RIS[in], structure];
};

ToRope: PUBLIC AC.ToRopeOp ~ { 
sequenceStructureData: SequenceStructureData _ NARROW[in.structure.instanceData];
elementStructure: AC.Structure _ sequenceStructureData.elementStructure;
inData: SequenceData _ NARROW[in.data];
out _ "( "; 
FOR i:NAT IN [1..inData.lengthPlus1) DO
out _ Rope.Concat[ out, elementStructure.class.toRope[inData[i] ] ];
    IF i < inData.lengthPlus1-1 THEN out _ Rope.Concat[out,", "];
ENDLOOP;
out _ Rope.Concat[ out, " )" ];
    };

Write: PUBLIC AC.WriteOp ~ {
stream.PutRope[ ToRope[in] ]
};  
<<Operations>>
Equal: PUBLIC AC.EqualityOp ~ {
data: SequenceStructureData _ NARROW[firstArg.structure.instanceData];
firstArgData: SequenceData _ NARROW[firstArg.data];
secondArgData: SequenceData _ NARROW[secondArg.data];
IF firstArgData.lengthPlus1 # secondArgData.lengthPlus1 THEN RETURN[FALSE];
FOR j: NAT IN [1..firstArgData.lengthPlus1) DO
IF NOT data.elementStructure.class.equal[firstArgData[j], secondArgData[j] ] THEN RETURN[FALSE];
ENDLOOP;
RETURN[TRUE];
};

    Sel: PUBLIC AC.BinaryOp ~ {
    firstData: SequenceData _ NARROW[firstArg.data];
    index: INT _ Ints.ToINT[secondArg];
    RETURN[firstData[index] ];
        };

    Ins: PUBLIC AC.TernaryOp ~ {
    sequenceStructureData: SequenceStructureData _ NARROW[firstArg.structure.instanceData];
    firstData: SequenceData _ NARROW[firstArg.data];
    insertAfter: INT _ Ints.ToINT[secondArg];
    newData: SequenceData _ NEW[SequenceDataRec[firstData.lengthPlus1] ];
    IF NOT sequenceStructureData.elementStructure.class.isElementOf[thirdArg, sequenceStructureData.elementStructure] THEN TypeError[];
    FOR i:INT IN [1..insertAfter] DO newData[i] _ firstData[i] ENDLOOP;
    newData[insertAfter+1] _ thirdArg;
    FOR j:INT IN [insertAfter+1..firstData.lengthPlus1) DO newData[j+1] _ firstData[j] ENDLOOP;
        RETURN[ NEW[AC.ObjectRec _ [
            structure: firstArg.structure,
            data: newData
            ] ] ];
        };

    Del: PUBLIC AC.BinaryOp ~ {
    sequenceStructureData: SequenceStructureData _ NARROW[firstArg.structure.instanceData];
    firstData: SequenceData _ NARROW[firstArg.data];
    delete: INT _ Ints.ToINT[secondArg];
    newData: SequenceData _ NEW[SequenceDataRec[firstData.lengthPlus1-2] ];
    FOR i:INT IN [1..delete-1] DO newData[i] _ firstData[i] ENDLOOP;
    FOR j:INT IN [delete+1..firstData.lengthPlus1) DO newData[j-1] _ firstData[j] ENDLOOP;
        RETURN[ NEW[AC.ObjectRec _ [
            structure: firstArg.structure,
            data: newData
            ] ] ];
        };

    ApplyUnOp: PUBLIC AC.BinaryMixedOp ~ {
    seqData: SequenceData _ NARROW[firstArg.data];
        refOp: REF AC.UnaryOp _ NARROW[secondArg];
    newElementStructure, newSeqStructure: AC.Structure _ NIL;
    newData: SequenceData _ NEW[SequenceDataRec[seqData.lengthPlus1-1] ];
    FOR i:INT IN [1..seqData.lengthPlus1) DO
    newData[i] _ refOp^[seqData[i]];
    IF newData[i]# NIL THEN newElementStructure _ newData[i].structure;
    ENDLOOP;
    IF newElementStructure = NIL THEN RETURN[NIL];
    newSeqStructure _ MakeSequenceStructure[newElementStructure];
        RETURN[ NEW[AC.ObjectRec _ [
            structure: newSeqStructure,
            data: newData
            ] ] ];
        };
