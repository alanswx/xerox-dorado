        elementStructure: AC.Structure
        ];

    SequenceOps: TYPE = REF SequenceOpsRec; -- prop key is $SequenceStructure.

        applyUnaryOpToElements: AC.BinaryMixedOp, 
        makeSequence: AC.ListImbedOp,
        firstNullaryOpName: Rope.ROPE _ NIL,
        firstNullaryOp: AC.NullaryOp _ NIL,
        secondNullaryOpName: Rope.ROPE _ NIL,
        secondNullaryOp: AC.NullaryOp _ NIL,
        firstUnaryOpName: Rope.ROPE _ NIL,
        firstUnaryOp: AC.UnaryOp _ NIL,
        secondUnaryOpName: Rope.ROPE _ NIL,
        secondUnaryOp: AC.UnaryOp _ NIL
        ];

    MakeSequenceStructure: PROC [elementStructure: AC.Structure, firstNullaryOpName: Rope.ROPE _ NIL, firstNullaryOp: AC.NullaryOp _ 
    NIL, firstUnaryOpName: Rope.ROPE _ NIL, firstUnaryOp: AC.UnaryOp _ NIL, secondNullaryOpName: Rope.ROPE _ NIL, secondNullaryOp: 
    AC.NullaryOp _ NIL, secondUnaryOpName: Rope.ROPE _ NIL, secondUnaryOp: AC.UnaryOp _ NIL] RETURNS [sequenceStructure:
