Cedar translation of the famous Dhrystone benchmark by Reinhold P. Weicker, CACM 27, No 10, 1984 p. 1013 ff. An effort has been made to preserve the ADA-ness of the original benchmark without overtuning the Cedar code.
NOTE: An effort has been made to preserve the intent behind the original ADA version of the benchmark and not to write a piece of supercharged Cedar code.
The salient Cedar modifications to the original code are:
1. RecordType is not written as a variant record
2. The use of Ropes rather than RefText or PACKED ARRAYs
3. The use of Rope.InlineFetch to index character strings
4. Func3 has been replaced by an INLINE
A few more modifcations by RRA are:
0. A little different formatting.
1. Timing is done more precisely (using BasicTime.Pulses).
2. 'iterations' kept as a CARDINAL instead of a CARD
3. Simulate out parameters & in/out parameters by explicit returns
4. replace use of ROPE by REF TEXT
5. no use of INLINE
A table of comparison benchmarks can be found in DhrystoneTable.tioga. The timing mechanism and the initiating routine, PerformBenchmark, are located at the end of this source.
The following is from the original program commentary:-

* The following program contains statements of a high-level programming
* language ADA in a distribution considered representative:
*
* assignments  53%
* control statements  32%
* procedure, function calls 15%
*
* 100 statements are dynamically executed. The program is balanced with
* respect to the three aspects:
*  - statement type
*  - operand type (for simple data types)
*  - operand access
*   operand global, local, parameter, or constant.
*
* The combination of these three aspects is balanced only approximately.
* The program does not compute anything meaningful, but it is
* syntactically and semantically correct.

CedarDhrystone: CEDAR PROGRAM
IMPORTS BasicTime, CedarProcess, Commander, IO, SystemVersion
= BEGIN
Types
Enumeration: TYPE = {ident1, ident2, ident3, ident4, ident5};
OneToThirty: TYPE = CARDINAL[0..30);
OneToFifty: TYPE = CARDINAL[0..50);
Array1DimInteger: TYPE = ARRAY OneToFifty OF INTEGER;
Array2DimInteger: TYPE = ARRAY OneToFifty OF ARRAY OneToFifty OF INTEGER;
CapitalLetter: TYPE = CHAR['A..'Z];
String30: TYPE = REF TEXT;
RecordPointer: TYPE = REF RecordType;
RecordType: TYPE = RECORD [
pointerComp: RecordPointer,
discr: Enumeration,
enumComp: Enumeration,
intComp: OneToFifty,
stringComp: String30
];
Globals
Timing variables
magicNumber: CARDINAL = 50000; --Recommended for 16-bit machines
iterations: CARDINAL ← magicNumber;
tStart: BasicTime.Pulses;
tareTime, tElapsed: REAL;
Benchmark variables
boolGlob: BOOL;
intGlob: INTEGER;
charGlob1, charGlob2: CHAR;
arrayGlob1: REF Array1DimInteger ← NEW[Array1DimInteger];
arrayGlob2: REF Array2DimInteger ← NEW[Array2DimInteger];
pointerGlob, pointerGlobNext: RecordPointer;
Dhrystone Benchmark Routines
DhryProcess: PROC = {
All other benchmark procedures are called by this one
intLoc1, intLoc2, intLoc3: OneToFifty;
charLoc: CHAR;  --Cedar compiler flags unused declarations
enumLoc: Enumeration;
stringLoc1, stringLoc2: String30;
pointerGlobNext ← NEW[RecordType];
pointerGlob ← NEW[RecordType ← [
pointerComp: pointerGlobNext,
discr: ident1,
enumComp: ident3,
intComp: 40,
stringComp: "DHRYSTONE PROGRAM, SOME STRING"]
];
stringLoc1 ← "DHRYSTONE PROGRAM, 1'ST STRING"; --30 chars
StartTimer[];
THROUGH [0..iterations) DO -- Test loop
Proc5[];
Proc4[];
charGlob1 = 'A, charGlob2 = 'B & boolGlob = FALSE
intLoc1 ← 2;
intLoc2 ← 3;
stringLoc2 ← "DHRYSTONE PROGRAM, 2'ND STRING"; --30 chars
enumLoc ← ident2;
boolGlob ← NOT Func2[stringLoc1, stringLoc2]; --TRUE
WHILE intLoc1 < intLoc2 DO --once
intLoc3 ← 5 + intLoc1 - intLoc2; --intLoc3 = 7
intLoc3 ← Proc7[intLoc1, intLoc2];
intLoc1 ← intLoc1.SUCC;
ENDLOOP;
--intLoc1 = 3
[arrayGlob1, arrayGlob2] ← Proc8[arrayGlob1, arrayGlob2, intLoc1, intLoc3];
--intGlob = 5
Proc1[pointerGlob];
FOR charIndex: CHAR IN ['A..charGlob2] DO --twice
IF enumLoc = Func1[charIndex, 'C] THEN enumLoc ← Proc6[ident1]; --not executed
ENDLOOP;
--enumLoc = ident1
--intLoc1 = 3, intLoc2 = 3, intLoc3 = 7
intLoc3 ← intLoc2 * intLoc1;
intLoc2 ← intLoc3 / intLoc1;
intLoc2 ← 7 * (intLoc3 - intLoc2) - intLoc1;
intLoc1 ← Proc2[intLoc1];
ENDLOOP; --over test iterations
StopTimer[];
};
Proc1: PROC [pointerParIn: RecordPointer] = {
Called once
OPEN nextRecord: pointerParIn.pointerComp^; -- RRA: alias, not new variable
nextRecord: RecordType ← pointerParIn.pointerComp^;
nextRecord ← pointerGlob^; -- RRA: added on December 16, 1986
pointerParIn.intComp ← 5;
nextRecord.intComp ← pointerParIn.intComp;
nextRecord.pointerComp ← pointerParIn.pointerComp;
nextRecord.pointerComp ← Proc3[];
IF nextRecord.discr = ident1
THEN {
nextRecord.intComp ← 6;
nextRecord.enumComp ← Proc6[pointerParIn.enumComp];
nextRecord.pointerComp ← pointerGlob.pointerComp;
nextRecord.intComp ← Proc7[nextRecord.intComp, 10];
}
ELSE --not executed
pointerParIn^ ← nextRecord;
};
Proc2: PROC [intParInOut: OneToFifty] RETURNS [OneToFifty] = {
Called once
intLoc: OneToFifty ← intParInOut + 10;
enumLoc: Enumeration;
DO
IF charGlob1 = 'A THEN {
intLoc ← intLoc.PRED;
intParInOut ← intLoc - intGlob;
enumLoc ← ident1; --TRUE
};
IF enumLoc = ident1 THEN EXIT;
ENDLOOP;
RETURN [intParInOut]; -- simulated in/out parameter
};
Proc3: PROC RETURNS [pointerParOut: RecordPointer] = {
Called once
IF pointerGlob # NIL
THEN pointerParOut ← pointerGlob.pointerComp
ELSE intGlob ← 100; -- not executed
pointerGlob.intComp ← Proc7[10, intGlob];
RETURN [pointerParOut]; -- simulated out parameter
};
Proc4: PROC = {
Called once
boolLoc: BOOL ← charGlob1 = 'A;
boolLoc ← boolLoc OR boolGlob;
charGlob2 ← 'B;
};
Proc5: PROC = {
Called once
charGlob1 ← 'A;
boolGlob ← FALSE;
};
Proc6: PROC [enumParIn: Enumeration] RETURNS [enumParOut: Enumeration] = {
Called once
enumParOut ← enumParIn;
IF NOT Func3[enumParIn] THEN enumParOut ← ident4;
SELECT enumParIn FROM
ident1 => enumParOut ← ident1;
ident2 => enumParOut ← IF intGlob > 100 THEN ident1 ELSE ident4;
ident3 => enumParOut ← ident2; --executed
ident4 => NULL;
ident5 => enumParOut ← ident3;
ENDCASE => NULL;
RETURN [enumParOut]; -- simulated out parameter
};
Proc7: PROC [intParIn1, intParIn2: OneToFifty] RETURNS [intParOut: OneToFifty] = {
Called thrice
intLoc: OneToFifty ← intParIn1 + 2;
intParOut ← intParIn2 + intLoc;
RETURN [intParOut]; -- simulated out parameter
};
Proc8: PROC [arrayParInOut1: REF Array1DimInteger,
arrayParInOut2: REF Array2DimInteger,
inParIn1, inParIn2: INTEGER]
RETURNS [REF Array1DimInteger, REF Array2DimInteger] = {
Called once
intLoc: OneToFifty ← inParIn1 + 5;
arrayParInOut1[intLoc] ← inParIn2;
arrayParInOut1[intLoc.SUCC] ← arrayParInOut1[intLoc];
arrayParInOut1[intLoc+30] ← intLoc;
FOR intIndex: OneToFifty IN [intLoc..intLoc.SUCC] DO --twice
arrayParInOut2[intLoc][intIndex] ← intLoc;
ENDLOOP;
arrayParInOut2[intLoc][intLoc.PRED] ← arrayParInOut2[intLoc][intLoc.PRED] + 1;
arrayParInOut2[intLoc+20][intLoc] ← arrayParInOut1[intLoc];
intGlob ← 5;
RETURN [arrayParInOut1, arrayParInOut2]; -- simulated in/out parameter
};
Func1: PROC [charParIn1, charParIn2: CapitalLetter] RETURNS [Enumeration] = {
Called thrice
charLoc1: CapitalLetter ← charParIn1;
charLoc2: CapitalLetter ← charLoc1;
RETURN [IF charLoc2 # charParIn2 THEN ident1 ELSE ident2];
};
Func2: PROC [stringParIn1, stringParIn2: String30] RETURNS [BOOL] = {
Called once
intLoc: OneToFifty ← 2;
charLoc: CapitalLetter;
WHILE intLoc <= 2 DO --once
IF Func1[stringParIn1[intLoc], stringParIn2[intLoc.SUCC]] = ident1 THEN {
charLoc ← 'A;
intLoc ← intLoc.SUCC;
};
ENDLOOP;
IF charLoc >= 'W AND charLoc <= 'Z THEN intLoc ← 7; --not executed
IF charLoc = 'X
THEN RETURN [TRUE] --not executed
ELSE --executed
IF StringCompare[stringParIn1, stringParIn2] = greater
THEN {
intLoc ← intLoc + 7;
RETURN [TRUE];
}
ELSE
RETURN [FALSE];
};
Func3: PROC [enumParIn1: Enumeration] RETURNS [BOOL] = {
enumLoc: Enumeration ← enumParIn1;
IF enumLoc = ident3 THEN RETURN [TRUE];
RETURN [FALSE];
};
Comparison: TYPE = {less, equal, greater};
StringCompare: PROC [r1, r2: String30] RETURNS [Comparison] = TRUSTED {
min: CARDINAL ← MIN[r1.length, r2.length];
pos: CARDINAL ← 0;
p1: LONG POINTER TO WORD ← LOOPHOLE[r1, LONG POINTER TO WORD] + SIZE[TEXT[0]];
p2: LONG POINTER TO WORD ← LOOPHOLE[r2, LONG POINTER TO WORD] + SIZE[TEXT[0]];
WHILE pos+1 < min DO
Fast loop (present in system routines)
IF p1^ # p2^ THEN EXIT;
p1 ← p1 + 1;
p2 ← p2 + 1;
pos ← pos + 2;
ENDLOOP;
FOR i: CARDINAL IN [pos..min) DO
c1: CHAR ← r1[i];
c2: CHAR ← r2[i];
IF c1 = c2 THEN LOOP;
IF c1 < c2 THEN RETURN [less] ELSE RETURN [greater];
ENDLOOP;
IF min < r1.length THEN RETURN [greater];
IF min < r2.length THEN RETURN [less];
RETURN [equal];
};
Timing Routines
CalibrateTimer: PROC [out: IO.STREAM] = {
IO.PutRope[out, "\nCalibrating... "];
StartTimer[];
THROUGH [0..iterations) DO --Nothing-- ENDLOOP;
StopTimer[];
tareTime ← tElapsed;
IO.PutRope[out, "done."];
};
StartTimer: PROC = {
tStart ← BasicTime.GetClockPulses[];
};
StopTimer: PROC = {
tElapsed ← BasicTime.PulsesToSeconds[BasicTime.GetClockPulses[] - tStart];
};
Benchmark execution commences here
PerformBenchmark: Commander.CommandProc = {
[cmd: Commander.Handle] RETURNS [result: REF ANY ← NIL, msg: ROPE ← NIL]
out: IO.STREAM ← cmd.out;
oldPriority: CedarProcess.Priority ← CedarProcess.GetPriority[];
IF cmd.procData.clientData = $Fast THEN CedarProcess.SetPriority[excited];
CalibrateTimer[out];
IO.PutRope[out, "\nExecuting..."];
DhryProcess[]; --Timer actions are embedded in DhryProcess
IF cmd.procData.clientData = $Fast THEN CedarProcess.SetPriority[oldPriority];
IO.PutRope[out, " done."];
Report the bad news
IO.PutF[out, "\nExecuted %g iterations in %4.2f Sec.\nThis %g rates as a %4.1f Dhrystones machine.\n", IO.card[iterations], [real[tElapsed-tareTime]],
IO.rope[SELECT SystemVersion.machineType FROM
dolphin => "Dolphin",
dorado => "Dorado",
dandelion => "Dandelion",
ENDCASE => "Dicentra"],
[real[iterations/tElapsed]]
];
};
Commander.Register[key: "CedarDhrystone", proc: PerformBenchmark,
doc: "Cedar version of Dhrystone benchmark."];
Commander.Register[key: "FastCedarDhrystone", proc: PerformBenchmark,
doc: "Cedar version of Dhrystone benchmark.",
clientData: $Fast];
