MARTIN RICHARDS' BENCHMARK PROGRAM transcribed from BCPL
via Modula version by Peter Robinson of Fri May 20 14:55:00 1983
via Mesa version by Jon Gibbons of Aug 5, 1983 22:41
into Cedar version by Dan Craft on Sept 9, 1983

Dr. Martin Richards of the Computer Laboratory, University of Cambridge, England has written a benchmark program for examining the performance of language/compiler/machine combinations in executing systems software. The program emulates the action of a small ("Tripos" like) operating system in scheduling and executing six tasks: two devices, two device handlers, a working task, and an idle task. The input device is driven by a pseudo-random stream of numbers, and the program terminates after a given amount of data has been processed. The total time taken to execute the program, and the resulting code size are measured.

The program was originally written in BCPL, and was intended as a measure not only of a machine's performance, but also of the quality of its BCPL implementation. Unfortunately, the only specification of the benchmark is the BCPL source code, thus restricting the implementation approach which might be taken with other languages. The benchmark has now been transcribed into a number of different languages, giving useful comparisons between their efficiency for systems programming.

Commander USING [CommandProc, Register],
ShowTime USING [GetMark, ShowDelta, Microseconds], -- all of the time measurement code is
commented out when measuring code size
IO USING [Put, PutF, PutChar, string, card, NewLine, STREAM],
Heap USING [CreateMDS, DeleteMDS],
Inline USING [BITAND, BITXOR];
IBenchMark: PROGRAM
IMPORTS Commander, IO, ShowTime, Heap, Inline =
BEGIN
z: MDSZone; -- heap to use for packets and tasks
in, out: IO.STREAM; -- only "out" used
tracing: BOOLEAN;
layout: CARDINAL;
charsPerLine: CARDINAL = 50;
Trace: PROCEDURE [ch: CHARACTER] = BEGIN
layout ← layout-1;
IF layout <= 0 THEN {out.NewLine[]; layout ← charsPerLine};
out.PutChar[ch];
END;
InitTrace: PROCEDURE [] = {tracing ← FALSE; layout ← 0};
TaskIdentity: TYPE = MACHINE DEPENDENT {idler(1), worker, handlerA, handlerB, deviceA, deviceB};
Task: TYPE = POINTER TO TaskControlBlock;
TaskControlBlock: TYPE = RECORD [
link: Task,
identity: TaskIdentity,
priority: TaskPriority,
input: WorkQueue,
state: TaskState,
function: TaskOperation,
handle: UNSPECIFIED
];
TaskPriority: TYPE = CARDINAL;
TaskState: TYPE = RECORD [
packetPending, taskWaiting, taskHolding: BOOLEAN ← FALSE];
running: TaskState = [];
waiting: TaskState = [taskWaiting: TRUE];
waitingWithPacket: TaskState = [taskWaiting: TRUE, packetPending: TRUE
];
TaskOperation: TYPE = PROCEDURE [work: WorkQueue, word: UNSPECIFIED] RETURNS [t: Task];
noTask: Task = NIL;
WorkQueue: TYPE = POINTER TO Packet;
Packet: TYPE = RECORD [
link: WorkQueue,
identity: TaskIdentity,
kind: PacketKind,
datum: UNSPECIFIED,
data: ARRAY [0..packetBufferSize] OF UNSPECIFIED
];
PacketKind: TYPE = {devicePacketKind, workPacketKind};
packetBufferSize: CARDINAL = 3;
noWork: WorkQueue = NIL;
queuePacketCount, holdCount: CARDINAL;
taskTable: ARRAY TaskIdentity OF Task;
taskList: Task;
currentTask: Task;
currentTaskIdentity: TaskIdentity;
InitScheduler: PROCEDURE [] = BEGIN
queuePacketCount ← holdCount ← 0;
taskTable ← ALL[noTask];
taskList ← noTask;
END;
CreateTask: PROCEDURE [identity: TaskIdentity, priority: TaskPriority, initialWorkQueue: WorkQueue, initialState: TaskState, function: TaskOperation, privateData: UNSPECIFIED] = BEGIN
t: Task ← z.NEW[TaskControlBlock ←
[link: taskList,
identity: identity,
priority: priority,
input: initialWorkQueue,
state: initialState,
function: function,
handle: privateData]];
taskList ← t;
taskTable[identity] ← t;
END;
CreatePacket: PROCEDURE [link: WorkQueue, identity: TaskIdentity, kind: PacketKind] RETURNS [workQueue: WorkQueue] = BEGIN
workQueue ← z.NEW[Packet ←
[link: link,
identity: identity,
kind: kind,
datum: 0,
data: ALL[0]]];
RETURN [workQueue];
END;
QueuePacket: PROCEDURE [packet: WorkQueue] RETURNS [task: Task] = INLINE BEGIN
t: Task ← FindTask[packet.identity];
IF t=noTask THEN RETURN [noTask];
queuePacketCount ← queuePacketCount+1;
packet.link ← noWork;
packet.identity ← currentTaskIdentity; -- ???? DHC
IF t.input=noWork THEN BEGIN
t.input ← packet;
t.state.packetPending ← TRUE;
IF t.priority > currentTask.priority THEN RETURN [t]
END
ELSE Append[packet, @t.input];
RETURN [currentTask];
END;
Append: PROCEDURE [packet: WorkQueue, head: POINTER TO WorkQueue] = INLINE BEGIN
packet.link ← noWork;
IF head^=noWork THEN {head^ ← packet; RETURN};
FOR mouse: WorkQueue ← head^, mouse.link UNTIL mouse.link=noWork DO
REPEAT
FINISHED => mouse.link ← packet
ENDLOOP;
END;
Schedule: PROCEDURE = BEGIN
message: WorkQueue;
currentTask ← taskList;
WHILE currentTask#noTask DO
message ← noWork;
IF currentTask.state.taskHolding OR currentTask.state=waiting THEN currentTask ← currentTask.link
ELSE BEGIN
IF currentTask.state=waitingWithPacket THEN BEGIN
message ← currentTask.input;
currentTask.input ← message.link;
IF currentTask.input=noWork THEN currentTask.state ← running
ELSE currentTask.state ← [packetPending: TRUE]
END;
currentTaskIdentity ← currentTask.identity;
IF tracing THEN Trace[LOOPHOLE[currentTaskIdentity, CARDINAL]+'0];
currentTask ← currentTask.function[message, currentTask.handle];
END;
ENDLOOP;
END;
Wait: PROCEDURE [] RETURNS [task: Task] = INLINE BEGIN
currentTask.state.taskWaiting ← TRUE;
RETURN [currentTask];
END;
HoldSelf: PROCEDURE [] RETURNS [task: Task] = INLINE BEGIN
holdCount ← holdCount+1;
currentTask.state.taskHolding ← TRUE;
RETURN [currentTask.link];
END;
Release: PROCEDURE [identity: TaskIdentity] RETURNS [task: Task] = INLINE BEGIN
t: Task ← FindTask[identity];
IF t=noTask THEN RETURN [noTask];
t.state.taskHolding ← FALSE;
IF t.priority > currentTask.priority THEN RETURN [t]
ELSE RETURN [currentTask];
END;
FindTask: PROCEDURE [identity: TaskIdentity] RETURNS [task: Task] = INLINE BEGIN
t: Task ← taskTable[identity];
IF t=noTask THEN ERROR;
RETURN [t];
END;
hashValue: CARDINAL = --0D008X-- 150010B; -- MOAN!!!!!
IdleTaskDataRecord: TYPE = RECORD [control: WORD, count: CARDINAL];
IdleTaskData: TYPE = POINTER TO IdleTaskDataRecord;
initialControl: WORD = 1;
CreateIdler: PROCEDURE [identity: TaskIdentity, priority: TaskPriority, work: WorkQueue, state: TaskState] = BEGIN
data: IdleTaskData ← z.NEW[IdleTaskDataRecord ← [initialControl, 10000]];
CreateTask[identity, priority, work, state, IdleFunction, data];
END;
IdleFunction: TaskOperation = BEGIN
data: IdleTaskData ← word;
data.count ← data.count-1;
IF data.count=0 THEN RETURN [HoldSelf[]];
IF Inline.BITAND[data.control, 1]=0 THEN {
data.control ← data.control/2;
RETURN [Release[deviceA]];
}
ELSE {data.control ← Inline.BITXOR[data.control/2, hashValue];
RETURN [Release[deviceB]]};
END;
DeviceTaskDataRecord: TYPE = RECORD [pending: WorkQueue];
DeviceTaskData: TYPE = POINTER TO DeviceTaskDataRecord;
CreateDevice: PROCEDURE [identity: TaskIdentity, priority: TaskPriority, work: WorkQueue, state: TaskState] =BEGIN
data: DeviceTaskData ← z.NEW[DeviceTaskDataRecord ← [noWork]];
CreateTask[identity, priority, work, state, DeviceFunction, data];
END;
DeviceFunction: TaskOperation = BEGIN
data: DeviceTaskData ← word;
IF work=noWork THEN BEGIN
IF data.pending=noWork THEN RETURN [Wait[]];
work ← data.pending;
data.pending ← noWork;
RETURN [QueuePacket[work]]
END
ELSE BEGIN
data.pending ← work;
IF tracing THEN Trace[work.datum];
RETURN [HoldSelf[]]
END;
END;
HandlerTaskDataRecord: TYPE = RECORD [workIn, deviceIn: WorkQueue];
HandlerTaskData: TYPE = POINTER TO HandlerTaskDataRecord;
CreateHandler: PROCEDURE [identity: TaskIdentity, priority: TaskPriority, work: WorkQueue, state: TaskState] = BEGIN
data: HandlerTaskData ← z.NEW[HandlerTaskDataRecord ← [noWork, noWork]];
CreateTask[identity, priority, work, state, HandlerFunction, data];
END;
HandlerFunction: TaskOperation = BEGIN
data: HandlerTaskData ← word;
workPacket, devicePacket: WorkQueue;
count: INTEGER;
IF work#noWork THEN {
IF work.kind=workPacketKind THEN Append[work, @data.workIn]
ELSE Append[work, @data.deviceIn];
};
IF data.workIn#noWork THEN BEGIN
workPacket ← data.workIn;
count ← workPacket.datum;
IF count > packetBufferSize THEN {
data.workIn ← data.workIn.link;
RETURN [QueuePacket[workPacket]];
};
IF data.deviceIn#noWork THEN BEGIN
devicePacket ← data.deviceIn;
data.deviceIn ← data.deviceIn.link;
devicePacket.datum ← workPacket.data[count];
workPacket.datum ← count+1;
RETURN [QueuePacket[devicePacket]]
END;
END;
RETURN [Wait[]];
END;
WorkerTaskDataRecord: TYPE = RECORD [destination: TaskIdentity, count: CARDINAL];
WorkerTaskData: TYPE = POINTER TO WorkerTaskDataRecord;
CreateWorker: PROCEDURE [identity: TaskIdentity, priority: TaskPriority, work: WorkQueue, state: TaskState] = BEGIN
data: WorkerTaskData ← z.NEW[WorkerTaskDataRecord ← [handlerA, 0]];
CreateTask[identity, priority, work, state, WorkFunction, data];
END;
WorkFunction: TaskOperation = BEGIN
data: WorkerTaskData ← word;
IF work=noWork THEN RETURN [Wait[]]
ELSE BEGIN
data.destination ← (IF data.destination=handlerA THEN handlerB ELSE handlerA);
work.identity ← data.destination;
work.datum ← 0;
FOR i: CARDINAL IN [0..packetBufferSize] DO
data.count ← data.count+1;
IF data.count > 26 THEN data.count ← 1;
work.data[i] ← 'A+data.count-1;
ENDLOOP;
RETURN [QueuePacket[work]];
END;
END;
Main: Commander.CommandProc = TRUSTED BEGIN
workQ: WorkQueue;
mark1, mark2, mark3, mark4: ShowTime.Microseconds; -- for benchmark timing
PutDigit: SAFE PROCEDURE [c: CHAR] RETURNS [BOOL] = TRUSTED BEGIN
out.PutChar[c];
RETURN [FALSE];
END;
mark1 ← ShowTime.GetMark[];
z ← Heap.CreateMDS[initial: 2];
in ← cmd.in;
out ← cmd.out;
out.Put[IO.string["Bench mark starting\n"]];
InitTrace[];
InitScheduler[];
CreateIdler[idler, 0, noWork, running];
workQ ← CreatePacket[noWork, worker, workPacketKind];
workQ ← CreatePacket[workQ, worker, workPacketKind];
CreateWorker[worker, 1000, workQ, waitingWithPacket];
workQ ← CreatePacket[noWork, deviceA, devicePacketKind];
workQ ← CreatePacket[workQ, deviceA, devicePacketKind];
workQ ← CreatePacket[workQ, deviceA, devicePacketKind];
CreateHandler[handlerA, 2000, workQ, waitingWithPacket];
workQ ← CreatePacket[noWork, deviceB, devicePacketKind];
workQ ← CreatePacket[workQ, deviceB, devicePacketKind];
workQ ← CreatePacket[workQ, deviceB, devicePacketKind];
CreateHandler[handlerB, 3000, workQ, waitingWithPacket];
CreateDevice[deviceA, 4000, noWork, waiting];
CreateDevice[deviceB, 5000, noWork, waiting];
out.Put[IO.string["Starting\n"]];
mark2 ← ShowTime.GetMark[];
Schedule[];
mark3 ← ShowTime.GetMark[];
out.Put[IO.string["Finished\n"]];
out.PutF["QueuePacket count = %g, HoldCount = %g\nThese results are ", IO.card[queuePacketCount], IO.card[holdCount]];
IF (queuePacketCount = 23246) AND (holdCount = 9297)
THEN out.Put[IO.string["correct"]]
ELSE out.Put[IO.string["wrong"]];
out.Put[IO.string["\nEnd of run\n"]];
mark4 ← ShowTime.GetMark[];
out.Put[IO.string["***** Scheduler time = "]];
ShowTime.ShowDelta[mark3 - mark2, PutDigit];
out.Put[IO.string[", Total time = "]];
ShowTime.ShowDelta[mark4 - mark1, PutDigit];
Heap.DeleteMDS[z];
END;
Commander.Register["IBenchMark", Main];
