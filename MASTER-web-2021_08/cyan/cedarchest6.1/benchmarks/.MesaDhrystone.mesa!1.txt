    <<Mesa translation of the famous Dhrystone benchmark by Reinhold P. Weicker,  CACM 27, No 10, 1984 p. 1013 ff.  An effort has been 
    made to preserve the ADA-ness of the original benchmark without overtuning the Cedar code.>>
    <<>>
    <<NOTE: An effort has been made to preserve the intent behind the original ADA version of the  benchmark and not to write a piece 
    of supercharged Cedar code.>>
        <<>>
    <<See CedarDhrystone.mesa for the original Mesa version.  This version uses LONG POINTER instead of REF.>>
    <<>>
        DIRECTORY
            BasicTime USING [GetClockPulses, Pulses, PulsesToSeconds],
            CedarProcess USING [GetPriority, Priority, SetPriority],
            Commander USING [CommandProc, Register],
            IO USING [card, PutF, PutRope, rope, STREAM],
            SystemVersion USING [machineType],
            UnsafeStorage USING [GetSystemUZone];
MesaDhrystone: PROGRAM 
    IMPORTS BasicTime, CedarProcess, Commander, IO, SystemVersion, UnsafeStorage
    = BEGIN
<<Types>>
    Enumeration: TYPE = {ident1, ident2, ident3, ident4, ident5};
    OneToThirty: TYPE = CARDINAL[0..30);
    OneToFifty: TYPE = CARDINAL[0..50);
    Array1DimInteger: TYPE = ARRAY OneToFifty OF INTEGER; 
    Array2DimInteger: TYPE = ARRAY OneToFifty OF ARRAY OneToFifty OF INTEGER;
    CapitalLetter: TYPE = CHAR['A..'Z];
    String30: TYPE = LONG STRING;

    RecordPointer: TYPE = LONG POINTER TO RecordType;
    RecordType: TYPE = RECORD [
        pointerComp: RecordPointer,
        discr: Enumeration,
        enumComp: Enumeration,
        intComp: OneToFifty,
        stringComp: String30
        ];
<<Globals>>
<<Timing variables>>
    magicNumber: CARDINAL = 50000; --Recommended for 16-bit machines
    iterations: CARDINAL _ magicNumber; 
    tStart: BasicTime.Pulses;
    tareTime, tElapsed: REAL;
<<Benchmark variables>>
    uz: UNCOUNTED ZONE _ UnsafeStorage.GetSystemUZone[];
    boolGlob: BOOL;
    intGlob: INTEGER;
    charGlob1, charGlob2: CHAR;
    arrayGlob1: LONG POINTER TO Array1DimInteger _ uz.NEW[Array1DimInteger];
    arrayGlob2: LONG POINTER TO Array2DimInteger _ uz.NEW[Array2DimInteger];
    pointerGlob, pointerGlobNext: RecordPointer; 
<<Dhrystone Benchmark Routines>>
    DhryProcess: PROC = { 
        <<All other benchmark procedures are called by this one>>
        intLoc1, intLoc2, intLoc3: OneToFifty;
        <<charLoc: CHAR;     --Cedar compiler flags unused declarations>>
        enumLoc: Enumeration;
        stringLoc1, stringLoc2: String30;

        pointerGlobNext _ uz.NEW[RecordType];
        pointerGlob _ uz.NEW[RecordType _ [
            pointerComp: pointerGlobNext,
            discr:  ident1,
            enumComp:  ident3,
            intComp:  40,
            stringComp:  "DHRYSTONE PROGRAM, SOME STRING"]
            ];

        stringLoc1 _ "DHRYSTONE PROGRAM, 1'ST STRING"; --30 chars

        StartTimer[];
        THROUGH [0..iterations) DO -- Test loop
            Proc5[];
            Proc4[]; 
            <<charGlob1 = 'A, charGlob2 = 'B & boolGlob = FALSE>>
            intLoc1 _ 2;
            intLoc2 _ 3;
            stringLoc2 _ "DHRYSTONE PROGRAM, 2'ND STRING"; --30 chars
            enumLoc _ ident2;
            boolGlob _ NOT Func2[stringLoc1, stringLoc2]; --TRUE
            WHILE intLoc1 < intLoc2 DO --once
                intLoc3 _ 5 + intLoc1 - intLoc2; --intLoc3 = 7
                intLoc3 _ Proc7[intLoc1, intLoc2];
                intLoc1 _ intLoc1.SUCC;
                ENDLOOP;
            <<--intLoc1 = 3>>
            [arrayGlob1, arrayGlob2] _ Proc8[arrayGlob1, arrayGlob2, intLoc1, intLoc3];
            <<--intGlob = 5>>
            Proc1[pointerGlob]; 
            FOR charIndex: CHAR IN ['A..charGlob2] DO --twice
                IF enumLoc = Func1[charIndex, 'C] THEN enumLoc _ Proc6[ident1]; --not executed
                ENDLOOP;
            <<--enumLoc = ident1>>
            <<--intLoc1 = 3, intLoc2 = 3, intLoc3 = 7>>
            intLoc3 _ intLoc2 * intLoc1;
            intLoc2 _ intLoc3 / intLoc1;
            intLoc2 _ 7 * (intLoc3 - intLoc2) - intLoc1;
            intLoc1 _ Proc2[intLoc1];
            ENDLOOP; --over test iterations
        StopTimer[];
        };

    Proc1: PROC [pointerParIn: RecordPointer] = {
        <<Called once>>
        OPEN nextRecord: pointerParIn.pointerComp^;  -- RRA: alias, not new variable
        <<nextRecord: RecordType _ pointerParIn.pointerComp^;>>
        nextRecord _ pointerGlob^;  -- RRA: added on December 16, 1986
        pointerParIn.intComp _ 5;
        nextRecord.intComp _ pointerParIn.intComp;
        nextRecord.pointerComp _ pointerParIn.pointerComp;
        nextRecord.pointerComp _ Proc3[];
        IF nextRecord.discr = ident1
            THEN {
                nextRecord.intComp _ 6;
                nextRecord.enumComp _ Proc6[pointerParIn.enumComp];
                nextRecord.pointerComp _ pointerGlob.pointerComp;
                nextRecord.intComp _ Proc7[nextRecord.intComp, 10];
                }
            ELSE --not executed
                pointerParIn^ _ nextRecord;
        };

    Proc2: PROC [intParInOut: OneToFifty] RETURNS [OneToFifty] = {
        <<Called once>>
        intLoc: OneToFifty _ intParInOut + 10;
        enumLoc: Enumeration;
        DO
            IF charGlob1 = 'A THEN {
                intLoc _ intLoc.PRED;
                intParInOut _ intLoc - intGlob;
                enumLoc _ ident1; --TRUE
                };
            IF enumLoc = ident1 THEN EXIT;
            ENDLOOP;
        RETURN [intParInOut];  -- simulated in/out parameter
        };

    Proc3: PROC RETURNS [pointerParOut: RecordPointer] = {
        <<Called once>>
        IF pointerGlob # NIL
            THEN pointerParOut _ pointerGlob.pointerComp
            ELSE intGlob _ 100; -- not executed
        pointerGlob.intComp _ Proc7[10, intGlob];
        RETURN [pointerParOut];  -- simulated out parameter
        };

    Proc4: PROC = {
        <<Called once>>
        boolLoc: BOOL _ charGlob1 = 'A;
        boolLoc _ boolLoc OR boolGlob;
        charGlob2 _ 'B;
        };

    Proc5: PROC = {
        <<Called once>>
        charGlob1 _ 'A;
        boolGlob _ FALSE;
        };

    Proc6: PROC [enumParIn: Enumeration] RETURNS [enumParOut: Enumeration] = {
        <<Called once>>
        enumParOut _ enumParIn;
        IF NOT Func3[enumParIn] THEN enumParOut _ ident4;
        SELECT enumParIn FROM
            ident1 => enumParOut _ ident1;
            ident2 => enumParOut _ IF intGlob > 100 THEN ident1 ELSE  ident4; 
            ident3 => enumParOut _ ident2; --executed
            ident4 => NULL;
            ident5 =>  enumParOut _ ident3;
            ENDCASE => NULL;
        RETURN [enumParOut];  -- simulated out parameter
        };

    Proc7: PROC [intParIn1, intParIn2: OneToFifty] RETURNS [intParOut: OneToFifty] = {
        <<Called thrice>>
        intLoc: OneToFifty _ intParIn1 + 2;
        intParOut _ intParIn2 + intLoc;
        RETURN [intParOut];  -- simulated out parameter
        };

    Proc8: PROC [arrayParInOut1: LONG POINTER TO Array1DimInteger,
    arrayParInOut2: LONG POINTER TO Array2DimInteger,
    inParIn1, inParIn2: INTEGER]
    RETURNS [LONG POINTER TO Array1DimInteger, LONG POINTER TO Array2DimInteger] = {
        <<Called once>>
        intLoc: OneToFifty _ inParIn1 + 5;
        arrayParInOut1[intLoc] _ inParIn2;
        arrayParInOut1[intLoc.SUCC] _ arrayParInOut1[intLoc];
        arrayParInOut1[intLoc+30] _ intLoc;
        FOR intIndex: OneToFifty IN [intLoc..intLoc.SUCC] DO --twice
            arrayParInOut2[intLoc][intIndex] _ intLoc;
            ENDLOOP;
        arrayParInOut2[intLoc][intLoc.PRED] _ arrayParInOut2[intLoc][intLoc.PRED] + 1;
        arrayParInOut2[intLoc+20][intLoc] _ arrayParInOut1[intLoc];
        intGlob _ 5;
        RETURN [arrayParInOut1, arrayParInOut2]; -- simulated in/out parameter
        };

    Func1: PROC [charParIn1, charParIn2: CapitalLetter] RETURNS [Enumeration] = {
        <<Called thrice>>
        charLoc1: CapitalLetter _ charParIn1;
        charLoc2: CapitalLetter _ charLoc1;
        RETURN [IF charLoc2 # charParIn2 THEN ident1 ELSE ident2];
        };

    Func2: PROC [stringParIn1, stringParIn2: String30] RETURNS [BOOL] = { 
        <<Called once>>
        intLoc: OneToFifty _ 2;
        charLoc: CapitalLetter;
        WHILE intLoc <= 2 DO --once
            IF Func1[stringParIn1[intLoc], stringParIn2[intLoc.SUCC]] = ident1 THEN {
                charLoc _ 'A;
                intLoc _ intLoc.SUCC;
                };
            ENDLOOP;
        IF charLoc >= 'W AND charLoc <= 'Z THEN intLoc _ 7; --not executed
        IF charLoc = 'X
            THEN RETURN [TRUE] --not executed
            ELSE  --executed
                IF StringCompare[stringParIn1, stringParIn2] = greater
                    THEN {
                        intLoc _ intLoc + 7;
                        RETURN [TRUE];
                        }
                    ELSE
                        RETURN [FALSE];
        };

    Func3: PROC [enumParIn1: Enumeration] RETURNS [BOOL] = { 
        enumLoc: Enumeration _ enumParIn1;
        IF enumLoc = ident3 THEN RETURN [TRUE];
        RETURN [FALSE];
        };

    Comparison: TYPE = {less, equal, greater};
    StringCompare: PROC [r1, r2: String30] RETURNS [Comparison] = {
        min: CARDINAL _ MIN[r1.length, r2.length];
        pos: CARDINAL _ 0;
        p1: LONG POINTER TO WORD _ LOOPHOLE[r1, LONG POINTER TO WORD] + SIZE[TEXT[0]];
        p2: LONG POINTER TO WORD _ LOOPHOLE[r2, LONG POINTER TO WORD] + SIZE[TEXT[0]];
        WHILE pos+1 < min DO
            <<Fast loop (present in system routines)>>
            IF p1^ # p2^ THEN EXIT;
            p1 _ p1 + 1;
            p2 _ p2 + 1;
            pos _ pos + 2;
            ENDLOOP;
        FOR i: CARDINAL IN [pos..min) DO
            c1: CHAR _ r1[i];
            c2: CHAR _ r2[i];
            IF c1 = c2 THEN LOOP;
            IF c1 < c2 THEN RETURN [less] ELSE RETURN [greater];
            ENDLOOP;
        IF min < r1.length THEN RETURN [greater];
        IF min < r2.length THEN RETURN [less];
        RETURN [equal];
        };
<<Timing Routines>>
    CalibrateTimer: PROC [out: IO.STREAM] = {
        IO.PutRope[out, "\nCalibrating... "];
        StartTimer[];
        THROUGH [0..iterations) DO --Nothing-- ENDLOOP;
        StopTimer[];
        tareTime _ tElapsed;
        IO.PutRope[out, "done."];
        };

    StartTimer: PROC = {
        tStart _ BasicTime.GetClockPulses[];
        };

    StopTimer: PROC = {
        tElapsed _ BasicTime.PulsesToSeconds[BasicTime.GetClockPulses[] - tStart];
        };
<<Benchmark execution commences here>>
    PerformBenchmark: Commander.CommandProc = TRUSTED {
        <<[cmd: Commander.Handle] RETURNS [result: REF ANY _ NIL, msg: ROPE _ NIL]>>
        out: IO.STREAM _ cmd.out;
        oldPriority: CedarProcess.Priority _ CedarProcess.GetPriority[];
        IF cmd.procData.clientData = $Fast THEN CedarProcess.SetPriority[excited];
        CalibrateTimer[out];
        IO.PutRope[out, "\nExecuting..."];
        DhryProcess[]; --Timer actions are embedded in DhryProcess
        IF cmd.procData.clientData = $Fast THEN CedarProcess.SetPriority[oldPriority];
        IO.PutRope[out, " done."];

        <<Report the bad news>>
        IO.PutF[out, "\nExecuted %g iterations in %4.2f Sec.\nThis %g rates as a %4.1f Dhrys/sec machine.\n", IO.card[iterations], 
        [real[tElapsed-tareTime]], 
            IO.rope[SELECT SystemVersion.machineType FROM
                dolphin => "Dolphin",
                dorado => "Dorado",
                dandelion => "Dandelion",
                ENDCASE => "Dicentra"], 
                [real[iterations/tElapsed]]
                ];

        };

    Commander.Register[key: "MesaDhrystone", proc: PerformBenchmark,
    doc: "Cedar version of Dhrystone benchmark."];
    Commander.Register[key: "FastMesaDhrystone", proc: PerformBenchmark,
    doc: "Cedar version of Dhrystone benchmark.",
    clientData: $Fast];

    END...
