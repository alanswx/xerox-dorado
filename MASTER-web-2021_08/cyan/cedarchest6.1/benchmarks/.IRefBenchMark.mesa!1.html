MARTIN RICHARDS' BENCHMARK PROGRAM transcribed from BCPL
via Modula version by Peter Robinson of Fri May 20 14:55:00 1983
via Mesa version by Jon Gibbons of Aug 5, 1983 22:41
into Cedar version by Dan Craft on Sept 9, 1983
Dr. Martin Richards of the Computer Laboratory, University of Cambridge, England has written a benchmark program for examining the performance of language/compiler/machine combinations in executing systems software. The program emulates the action of a small ("Tripos" like) operating system in scheduling and executing six tasks: two devices, two device handlers, a working task, and an idle task. The input device is driven by a pseudo-random stream of numbers, and the program terminates after a given amount of data has been processed. The total time taken to execute the program, and the resulting code size are measured.
The program was originally written in BCPL, and was intended as a measure not only of a machine's performance, but also of the quality of its BCPL implementation. Unfortunately, the only specification of the benchmark is the BCPL source code, thus restricting the implementation approach which might be taken with other languages. The benchmark has now been transcribed into a number of different languages, giving useful comparisons between their efficiency for systems programming.

IRefBenchMark: PROGRAM IMPORTS Commander, IO, BasicTime, Basics =
BEGIN
in, out: IO.STREAM; -- only "out" used
charsPerLine: CARDINAL = 50;
tracing: BOOLEAN;
layout: CARDINAL;
Trace: PROC [ch: CHARACTER] = BEGIN
layout ← layout-1;
IF layout <= 0
THEN {out.PutRope["\n"]; layout ← charsPerLine};
out.PutChar[ch];
END;
InitTrace: PROC = {tracing ← FALSE; layout ← 0};
TaskIdentity: TYPE = MACHINE DEPENDENT {idler(1), worker, handlerA, handlerB, deviceA, deviceB};
WorkQueue: TYPE = --Cedar--REF Packet;
Task: TYPE = --Cedar--REF TaskControlBlock;
TaskPriority: TYPE = CARDINAL;
TaskState: TYPE = RECORD [packetPending, taskWaiting, taskHolding: BOOLEAN ← FALSE];
running: TaskState = [];
waiting: TaskState = [taskWaiting: TRUE];
waitingWithPacket: TaskState = [taskWaiting: TRUE, packetPending: TRUE];
TaskOperation: TYPE = PROC [work: WorkQueue, taskArg: --Cedar--REF]
RETURNS [t: Task];
TaskControlBlock: TYPE = RECORD
[link: Task,
identity: TaskIdentity,
priority: TaskPriority,
input: WorkQueue,
state: TaskState,
function: TaskOperation,
handle: --Cedar--REF ];
noTask: Task = NIL;
PacketKind: TYPE = {devicePacketKind, workPacketKind};
packetBufferSize: CARDINAL = 3;
Packet: TYPE = RECORD
[link: WorkQueue,
identity: TaskIdentity,
kind: PacketKind,
datum: UNSPECIFIED,
data: ARRAY [0..packetBufferSize] OF UNSPECIFIED];
noWork: WorkQueue = NIL;
queuePacketCount, holdCount: CARDINAL;
taskTable: ARRAY TaskIdentity OF Task;
taskList: Task;
currentTask: Task;
currentTaskIdentity: TaskIdentity;
CreateTask: PROC
[identity: TaskIdentity, priority: TaskPriority,
initialWorkQueue: WorkQueue, initialState: TaskState,
function: TaskOperation, privateData: --Cedar--REF] =
BEGIN
t: Task ← NEW[TaskControlBlock ←
[link: taskList,
identity: identity,
priority: priority,
input: initialWorkQueue,
state: initialState,
function: function,
handle: privateData]];
taskList ← t;
taskTable[identity] ← t;
END;
CreatePacket: PROC
[link: WorkQueue, identity: TaskIdentity, kind: PacketKind]
RETURNS [p: WorkQueue] =
BEGIN
p ← NEW[Packet ←
[link: link, identity: identity, kind: kind, datum: 0, data: ALL[0]]];
END;
QueuePacket: PROC [packet: WorkQueue] RETURNS [t: Task] = INLINE
BEGIN
IF (t ← FindTask[packet.identity])=noTask THEN RETURN;
queuePacketCount ← queuePacketCount+1;
packet.link ← noWork;
packet.identity ← currentTaskIdentity; -- ???? DHC
IF t.input=noWork
THEN {
t.input ← packet;
t.state.packetPending ← TRUE;
IF t.priority > currentTask.priority THEN RETURN}
ELSE
t.input ← Append[packet, t.input];
RETURN [currentTask];
END;
Append: PROC [packet: WorkQueue, head: WorkQueue] RETURNS[newHead: WorkQueue] = INLINE
BEGIN --Cedar
packet.link ← noWork;
IF head=noWork THEN RETURN[packet]; --Cedar
FOR mouse: WorkQueue ← head, mouse.link WHILE mouse.link#noWork DO
REPEAT
FINISHED => {mouse.link ← packet; RETURN[head]};
ENDLOOP;
END;
Schedule: PROC = BEGIN
message: WorkQueue;
currentTask ← taskList;
WHILE currentTask#noTask DO
message ← noWork;
IF currentTask.state.taskHolding OR currentTask.state=waiting
THEN
currentTask ← currentTask.link
ELSE {
IF currentTask.state=waitingWithPacket
THEN {
message ← currentTask.input;
currentTask.input ← message.link;
currentTask.state ← IF currentTask.input=noWork
THEN running ELSE [packetPending: TRUE]};
currentTaskIdentity ← currentTask.identity;
IF tracing THEN
Trace[LOOPHOLE[currentTaskIdentity, CARDINAL]+'0];
currentTask ← currentTask.function[message, currentTask.handle]};
ENDLOOP;
END;
Wait: PROC [] RETURNS [Task] = INLINE {
currentTask.state.taskWaiting ← TRUE;
RETURN [currentTask]};
HoldSelf: PROC [] RETURNS [Task] = INLINE {
holdCount ← holdCount+1;
currentTask.state.taskHolding ← TRUE;
RETURN [currentTask.link]};
Release: PROC [identity: TaskIdentity] RETURNS [t: Task] = INLINE {
t ← FindTask[identity];
IF t=noTask THEN RETURN;
t.state.taskHolding ← FALSE;
RETURN[IF t.priority > currentTask.priority THEN t ELSE currentTask]};
FindTask: PROC [identity: TaskIdentity] RETURNS [t: Task] = INLINE {
IF (t ← taskTable[identity])=noTask THEN ERROR};
InitScheduler: PROC = {
queuePacketCount ← holdCount ← 0;
taskTable ← ALL[noTask];
taskList ← noTask};
initialControl: WORD = 1;
hashValue: CARDINAL = --0D008X-- 150010B; -- MOAN!!!!!
IdleTaskDataRecord: TYPE =
RECORD [control: WORD, count: CARDINAL];
IdleTaskData: TYPE = --Cedar--REF IdleTaskDataRecord;
CreateIdler: PROC [identity: TaskIdentity, priority: TaskPriority,
work: WorkQueue, state: TaskState] =
BEGIN
data: IdleTaskData ← NEW[IdleTaskDataRecord ← [initialControl, 10000]];
CreateTask[identity, priority, work, state, IdleFunction, data];
END;
IdleFunction: TaskOperation =
BEGIN
data: IdleTaskData ← NARROW[taskArg];
data.count ← data.count-1;
IF data.count=0 THEN RETURN [HoldSelf[]];
IF Basics.BITAND[data.control, 1]=0 THEN {
data.control ← data.control/2; RETURN [Release[deviceA]]}
ELSE {
data.control ← Basics.BITXOR[data.control/2, hashValue];
RETURN [Release[deviceB]]};
END;

};
DeviceTaskData: TYPE = --Cedar--REF DeviceTaskDataRecord;
DeviceTaskDataRecord: TYPE = RECORD [pending: WorkQueue];
CreateDevice: PROC [identity: TaskIdentity, priority: TaskPriority,
work: WorkQueue, state: TaskState] = {
data: DeviceTaskData ← NEW[DeviceTaskDataRecord ← [pending: noWork]];
CreateTask[identity, priority, work, state, DeviceFunction, data]};
DeviceFunction: TaskOperation =
BEGIN
data: DeviceTaskData = NARROW[taskArg];
IF work=noWork THEN {
IF data.pending=noWork THEN RETURN [Wait[]];
work ← data.pending;
data.pending ← noWork;
RETURN [QueuePacket[work]]}
ELSE {
data.pending ← work;
IF tracing THEN Trace[work.datum];
RETURN [HoldSelf[]]};
END;
HandlerTaskData: TYPE = --Cedar--REF HandlerTaskDataRecord;
HandlerTaskDataRecord: TYPE = RECORD [workIn, deviceIn: WorkQueue];
CreateHandler: PROC [identity: TaskIdentity, priority: TaskPriority,
work: WorkQueue, state: TaskState] =
BEGIN
data: HandlerTaskData ← NEW[HandlerTaskDataRecord ← [noWork, noWork]];
CreateTask[identity, priority, work, state, HandlerFunction, data];
END;
HandlerFunction: TaskOperation =
BEGIN
data: HandlerTaskData = NARROW[taskArg];
workPacket, devicePacket: WorkQueue;
count: INTEGER;
IF work#noWork
THEN IF work.kind=workPacketKind
THEN data.workIn𡤊ppend[work, data.workIn]
ELSE data.deviceIn𡤊ppend[work, data.deviceIn];
IF data.workIn#noWork
THEN
BEGIN
workPacket ← data.workIn;
count ← workPacket.datum;
IF count > packetBufferSize
THEN {data.workIn ← data.workIn.link; RETURN [QueuePacket[workPacket]]};
IF data.deviceIn#noWork
THEN
BEGIN
devicePacket ← data.deviceIn;
data.deviceIn ← data.deviceIn.link;
devicePacket.datum ← workPacket.data[count];
workPacket.datum ← count+1;
RETURN [QueuePacket[devicePacket]]
END;
END;
RETURN [Wait[]];
END;
WorkerTaskData: TYPE = --Cedar--REF WorkerTaskDataRecord;
WorkerTaskDataRecord: TYPE = RECORD [destination: TaskIdentity, count: CARDINAL];
CreateWorker: PROC [identity: TaskIdentity, priority: TaskPriority,
work: WorkQueue, state: TaskState] =
BEGIN
data: WorkerTaskData ← NEW[WorkerTaskDataRecord ← [destination: handlerA, count: 0]];
CreateTask[identity, priority, work, state, WorkFunction, data];
END;
WorkFunction: TaskOperation =
BEGIN
data: WorkerTaskData ← NARROW[taskArg];
IF work=noWork THEN RETURN [Wait[]]
ELSE {
data.destination ←
(IF data.destination#handlerA THEN handlerA ELSE handlerB);
work.identity ← data.destination;
work.datum ← 0;
FOR i: CARDINAL IN [0..packetBufferSize]
DO
data.count ← data.count+1;
IF data.count > 26 THEN data.count ← 1;
work.data[i] ← 'A+data.count-1;
ENDLOOP;
RETURN [QueuePacket[work]]};
END;
Main: Commander.CommandProc = TRUSTED
BEGIN
workQ: WorkQueue;
mark1, mark2, mark3, mark4: LONG CARDINAL; -- for benchmark timing; really Microseconds
PutDigit: SAFE PROC [c: CHAR] RETURNS [BOOL] = TRUSTED
BEGIN out.PutChar[c]; RETURN [FALSE]; END;
mark1 ← BasicTime.PulsesToMicroseconds[BasicTime.GetClockPulses[]];
in ← cmd.in;
out ← cmd.out;
out.Put[IO.rope["Bench mark starting\n"]];
InitTrace[];
InitScheduler[];
CreateIdler[idler, 0, noWork, running];
workQ ← CreatePacket[noWork, worker, workPacketKind];
workQ ← CreatePacket[workQ, worker, workPacketKind];
CreateWorker[worker, 1000, workQ, waitingWithPacket];
workQ ← CreatePacket[noWork, deviceA, devicePacketKind];
workQ ← CreatePacket[workQ, deviceA, devicePacketKind];
workQ ← CreatePacket[workQ, deviceA, devicePacketKind];
CreateHandler[handlerA, 2000, workQ, waitingWithPacket];
workQ ← CreatePacket[noWork, deviceB, devicePacketKind];
workQ ← CreatePacket[workQ, deviceB, devicePacketKind];
workQ ← CreatePacket[workQ, deviceB, devicePacketKind];
CreateHandler[handlerB, 3000, workQ, waitingWithPacket];
CreateDevice[deviceA, 4000, noWork, waiting];
CreateDevice[deviceB, 5000, noWork, waiting];
out.Put[IO.rope["Starting\n"]];
mark2 ← BasicTime.PulsesToMicroseconds[BasicTime.GetClockPulses[]];
Schedule[];
mark3 ← BasicTime.PulsesToMicroseconds[BasicTime.GetClockPulses[]];
out.Put[IO.rope["Finished\n"]];
out.PutF["QueuePacket count = %g, HoldCount = %g\nThese results are ", IO.card[queuePacketCount], IO.card[holdCount]];
IF (queuePacketCount = 23246) AND (holdCount = 9297)
THEN out.Put[IO.rope["correct"]]
ELSE out.Put[IO.rope["wrong"]];
out.Put[IO.rope["\nEnd of run\n"]];
mark4 ← BasicTime.PulsesToMicroseconds[BasicTime.GetClockPulses[]];
out.Put[IO.rope["***** Scheduler time = "]];
out.PutF["%g", IO.card[mark3 - mark2]];
out.Put[IO.rope[", Total time = "]];
out.PutF["%g", IO.card[mark4 - mark1]];
END;
Commander.Register["IBenchMark", Main];
